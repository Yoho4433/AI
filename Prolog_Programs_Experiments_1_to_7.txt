EXPERIMENT 4 — RECURSION IN PROLOG
================================================

1. Recursive Ancestor Relationship
-----------------------------------
parent(john,mary).
parent(mary,alice).
parent(alice,bob).
parent(bob,tom).

ancestor(X,Y) :-
    parent(X,Y).

ancestor(X,Y) :-
    parent(X,Z),
    ancestor(Z,Y).

Query:
?- ancestor(john,tom).

2. Tower of Hanoi
-----------------
hanoi(1,S,D,_) :-
    write('Move disk from '),
    write(S),
    write(' to '),
    write(D), nl.

hanoi(N,S,D,A) :-
    N > 1,
    M is N-1,
    hanoi(M,S,A,D),
    hanoi(1,S,D,A),
    hanoi(M,A,D,S).

Query:
?- hanoi(3,left,right,middle).

3. Depth-First Search (DFS) with Cycle Detection
------------------------------------------------
edge(a,b).
edge(a,c).
edge(b,d).
edge(c,e).
edge(d,f).
edge(e,f).

dfs(Goal,Goal,Visited,[Goal|Visited]).

dfs(Start,Goal,Visited,Path) :-
    edge(Start,Next),
    \+ member(Next,Visited),
    dfs(Next,Goal,[Start|Visited],Path).

Query:
?- dfs(a,f,[],P).

4. Breadth-First Search (BFS)
-----------------------------
edge(a,b).
edge(a,c).
edge(b,d).
edge(c,e).
edge(d,f).
edge(e,g).

bfs([[Goal|Path]|_],Goal,[Goal|Path]).

bfs([[Node|Path]|Rest],Goal,FinalPath) :-
    findall([Next,Node|Path],
            edge(Node,Next),
            NewPaths),
    append(Rest,NewPaths,Queue),
    bfs(Queue,Goal,FinalPath).

Query:
?- bfs([[a]],g,P).

Note: The PDF also contains headings for other recursive search examples, but the corresponding code is incomplete or not legible in the extracted source.

EXPERIMENT 6 — STRING OPERATIONS IN PROLOG
================================================

1. Convert a String to a List of Characters
-------------------------------------------
?- string_chars("hello", List).

2. String Concatenation
-----------------------
concatenate(String1, String2, Result) :-
    string_concat(String1, String2, Result).

Query:
?- concatenate("Artificial", " Intelligence", Result).

3. String Reversal
------------------
reverse_string(String, Reversed) :-
    string_chars(String, Characters),
    reverse(Characters, RevCharacters),
    string_chars(Reversed, RevCharacters).

Query:
?- reverse_string("prolog", Result).

4. Palindrome Checking
----------------------
palindrome(String) :-
    string_chars(String, Characters),
    reverse(Characters, Reversed),
    Characters = Reversed.

Queries:
?- palindrome("madam").
?- palindrome("computer").

5. Character Position
---------------------
character_position(String, Character, Position) :-
    string_chars(String, Characters),
    nth1(Position, Characters, Character).

Query:
?- character_position("prolog", 'o', Position).

6. Substring Position
---------------------
substring_position(String, Substring, Position) :-
    sub_string(String, Before, _, _, Substring),
    Position is Before + 1.

Query:
?- substring_position("artificial intelligence", "intelligence", Position).

7. Substring Extraction
-----------------------
extract_substring(String, Start, Length, Substring) :-
    Before is Start - 1,
    sub_string(String, Before, Length, _, Substring).

Query:
?- extract_substring("programming", 4, 5, Result).

EXPERIMENT 7 — SET OPERATIONS IN PROLOG

1. Membership Predicate
-----------------------
member_of(X, [X|_]).

member_of(X, [_|T]) :-
    member_of(X, T).

Query:

2. Union
--------
union([], L, L).

union([H|T], L, R) :-
    member_of(H, L),
    union(T, L, R).

union([H|T], L, [H|R]) :-
    \+ member_of(H, L),
    union(T, L, R).

Query:
?- union([1,2,3], [3,4,5], R).

3. Intersection
---------------
intersection([], _, []).

intersection([H|T], L, [H|R]) :-
    member_of(H, L),
    intersection(T, L, R).

intersection([H|T], L, R) :-
    \+ member_of(H, L),
    intersection(T, L, R).

Query:
?- intersection([1,2,3,4], [3,4,5,6], R).

4. Difference
-------------
difference([], _, []).

difference([H|T], L, R) :-
    member_of(H, L),
    difference(T, L, R).

difference([H|T], L, [H|R]) :-
    \+ member_of(H, L),
    difference(T, L, R).

Query:
?- difference([1,2,3,4,5], [2,4,6], R).

5. Symmetric Difference
-----------------------
symmetric_difference(A, B, R) :-
    difference(A, B, D1),
    difference(B, A, D2),
    union(D1, D2, R).

Query:
?- symmetric_difference([1,2,3,4], [3,4,5,6], R).
